# Persistent Runtime Vertical Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a persistent, authenticated, loopback Runtime with durable sessions, runs, event replay, LangGraph SQLite checkpoints, and a read-only compatibility agent while preserving every existing CLI path.

**Architecture:** Application state lives in `app.sqlite`; LangGraph execution checkpoints live independently in `checkpoints.sqlite`. An async Runtime service coordinates repositories and a read-only agent runner, while a thin WebSocket adapter authenticates clients and exposes versioned commands, responses, events, replay, and snapshot recovery.

**Tech Stack:** Python 3.11, `aiosqlite==0.22.1`, `langgraph-checkpoint-sqlite==3.1.0`, `websockets==15.0.1`, Deep Agents 0.6.12, LangGraph 1.2.9, pytest 8.4.1, SQLite 3.50.

```python
from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True, slots=True)
class RuntimePaths:
    home: Path
    state_dir: Path
    app_db: Path
    checkpoints_db: Path
    lock_file: Path
    discovery_file: Path
    token_file: Path

    @classmethod
    def from_environment(cls) -> "RuntimePaths":
        override = os.getenv("EEE_RUNTIME_HOME")
        if override is not None:
            if not override.strip():
                raise ValueError("EEE_RUNTIME_HOME must not be empty")
            home = Path(override)
            if not home.is_absolute():
                raise ValueError("EEE_RUNTIME_HOME must be absolute")
        else:
            local = os.getenv("LOCALAPPDATA")
            if not local:
                raise ValueError("LOCALAPPDATA is required when EEE_RUNTIME_HOME is unset")
            home = Path(local) / "EEEAgent"
        home = home.resolve()
        if home.exists() and not home.is_dir():
            raise ValueError("Runtime home must be a directory")
        state = home / "state"
        return cls(
            home=home,
            state_dir=state,
            app_db=state / "app.sqlite",
            checkpoints_db=state / "checkpoints.sqlite",
            lock_file=state / "runtime.lock",
            discovery_file=state / "runtime.json",
            token_file=state / "runtime.token",
        )

    def create_used_directories(self) -> None:
        self.state_dir.mkdir(parents=True, exist_ok=True)
```

Use the verified async context-manager API:

```python
class CheckpointManager:
    def __init__(self, path: Path) -> None:
        self._path = path
        self._context = None
        self.saver: AsyncSqliteSaver | None = None

    async def __aenter__(self) -> "CheckpointManager":
        os.environ.setdefault("LANGGRAPH_STRICT_MSGPACK", "true")
        self._context = AsyncSqliteSaver.from_conn_string(str(self._path))
        self.saver = await self._context.__aenter__()
        await self.saver.setup()
        return self

    async def __aexit__(self, exc_type, exc, traceback) -> None:
        context, self._context = self._context, None
        self.saver = None
        if context is not None:
            await context.__aexit__(exc_type, exc, traceback)

    async def delete_thread(self, session_id: str) -> None:
        if self.saver is None:
            raise RuntimeError("checkpoint manager is not open")
        await self.saver.adelete_thread(session_id)
```
