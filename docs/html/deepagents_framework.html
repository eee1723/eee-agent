1. DeepAgents 核心组装逻辑 (`eee_agent/app.py`)

在本项目中，`build_agent()` 是创建 Agent 状态图的总入口。它调用 `deepagents.create_deep_agent(...)` 并注入白名单工具、系统提示词与多个自定义可靠性中间件：

z:\EEE_Project\EEEProceduralModeling\eee_agent\app.py Python 3.11
from __future__ import annotations import os from collections.abc import Sequence from deepagents import create_deep_agent from langchain_core.tools import BaseTool from langgraph.checkpoint.base import BaseCheckpointSaver from langgraph.graph.state import CompiledStateGraph from eee_agent.harness import configure_deepagents_harness from eee_agent.model import build_model from eee_agent.system_prompt import build_system_prompt def build_agent( *, tools: Sequence[BaseTool] | None = None, checkpointer: BaseCheckpointSaver | None = None, context_schema: type | None = None, ) -> CompiledStateGraph: if tools is None: raise TypeError("build_agent requires an explicit secure tools allowlist") model = build_model() # 实例化 LLM Provider (如 DeepSeek V4) middleware = [] # 1. 挂载历史读取剪枝中间件 from eee_agent import context_trim if context_trim.is_enabled(): middleware.append(context_trim.TrimReadbacksMiddleware()) # 2. 挂载死循环保护中间件 from eee_agent import loop_guard if loop_guard.is_enabled(): middleware.append(loop_guard.LoopGuardMiddleware()) # 3. 挂载报错追查中间件 from eee_agent import tool_error_trace middleware.append(tool_error_trace.ToolErrorTraceMiddleware()) # 4. 挂载按需对话压缩 StateBackend backend = None if os.getenv("EEE_COMPACT_TOOL", "true").strip().lower() == "true": from deepagents.backends import StateBackend from deepagents.middleware import create_summarization_tool_middleware backend = StateBackend() middleware.append(create_summarization_tool_middleware(model, backend)) selected_tools = list(tools) kwargs = dict(model=model, tools=selected_tools, system_prompt=build_system_prompt(), middleware=middleware) if backend is not None: kwargs["backend"] = backend if checkpointer is not None: kwargs["checkpointer"] = checkpointer configure_deepagents_harness() return create_deep_agent(**kwargs)
📌 原理解析：
`build_agent` 强制要求传入显式的 `tools` 允许列表（没有全局写工具）。`create_deep_agent` 内部会将 `model`、`tools`、`system_prompt` 和 `middleware` 链编译为一个 LangGraph 状态图 `CompiledStateGraph`。当挂载了 `checkpointer`（SQLite 保存器）时，Agent 的对话状态会自动进行持久化，支持掉线恢复与断点会话续传。
2. 死循环守护者真实代码 (`eee_agent/loop_guard.py`)

当 Agent 在建模排错时陷入死循环（如不断重复创建和删除同名节点），传统 LLM 无法自我觉察。`LoopGuardMiddleware` 通过提取工具调用的签名 `(tool, node, parm)`，在中间件中进行动态拦截：

z:\EEE_Project\EEEProceduralModeling\eee_agent\loop_guard.py Python 3.11
from langchain.agents.middleware.types import AgentMiddleware, ModelRequest, ModelResponse from langchain_core.messages import SystemMessage # 只监控带有修改性质的工具，读操作不视作死循环 MUTATIVE = frozenset({ "delete_node", "create_node", "set_vex", "set_parms", "set_expression", "cook_node", "connect_nodes", }) def _sig(name, args) -> Optional[Tuple[str, str, str]]: """把 Tool Call 压缩为 (tool, node_path, parm_name) 三元组""" if name not in MUTATIVE: return None args = args or {} node = (args.get("node_path") or args.get("name") or "") parm = args.get("parm") or "" return (str(name), str(node), str(parm)) class LoopGuardMiddleware(AgentMiddleware): @staticmethod def _directive(calls: List[Tuple[str, str, str]]) -> Optional[str]: if not calls: return None last = calls[-1] n = sum(1 for c in calls if c == last) # 统计最后一次调用的重复次数 soft, hard = 3, 5 if n >= hard: return f"[LOOP GUARD] You have called {last} {n} times. You MUST stop now!" if n >= soft: return f"[LOOP GUARD] You've called {last} {n} times. STOP retrying! Change approach." return None @override def wrap_model_call(self, request: ModelRequest, handler: Callable) -> ModelResponse: calls = _recent_calls(request.messages, limit=16) directive = self._directive(calls) if directive: # 动态将警告追加到 SystemMessage 尾部，强行矫正 LLM 的下一步决策 request = _append_system(request, directive) return handler(request)
📌 原理解析：
继承 `AgentMiddleware` 并重写 `wrap_model_call` 可以在请求被发送给大模型（如 DeepSeek）之前拦截 `ModelRequest`。通过检查历史 `request.messages` 中的 `tool_calls`，当发现同一个三元组 `(set_vex, /obj/geo1/attribwrangle1, "")` 被连续触发 **3 次**时，自动向系统的 `SystemMessage` 强行追加一条指令，告诫 LLM“必须停止该方案，更换思路！”；达到 **5 次**时强制要求停下输出最终总结，彻底消灭死循环！
3. 上下文结果剪枝真实代码 (`eee_agent/context_trim.py`)

在 3D 建模过程中，查询节点状态 `work_status` 或几何体统计 `geometry_stats` 会返回海量 JSON。若不处理，多轮对话下来仅 Tool Response 就能消耗上百万 Token。`TrimReadbacksMiddleware` 优雅地解决了这个问题：

z:\EEE_Project\EEEProceduralModeling\eee_agent\context_trim.py Python 3.11
from langchain_core.messages import ToolMessage, AnyMessage READBACK_TOOLS = frozenset({ "work_status", "geometry_stats", "anchor_graph", "describe_node_type", "hou_status", "validate_geometry", }) STUB = "[older result omitted — re-call the tool to see the current state]" def _stub(message: AnyMessage) -> AnyMessage: """复制一份 ToolMessage，将其庞大的 content 内容替换为一句话 Stub""" if not isinstance(message, ToolMessage): return message return ToolMessage( content=STUB, tool_call_id=getattr(message, "tool_call_id", ""), name=getattr(message, "name", ""), ) class TrimReadbacksMiddleware(AgentMiddleware): @staticmethod def _trim(messages: List[AnyMessage]) -> Tuple[List[AnyMessage], bool]: last_idx = {} # 1. 扫描消息记录，找到每个 Read-back 工具最后一次调用的索引位置 for i, m in enumerate(messages): nm = getattr(m, "name", None) if isinstance(nm, "str") and nm in READBACK_TOOLS: last_idx[nm] = i # 2. 保留最后一次的完整 JSON，将旧的同名 ToolMessage 全部替换为 Stub 存根 out = list(messages) changed = False for i, m in enumerate(out): nm = getattr(m, "name", None) if nm in READBACK_TOOLS and last_idx.get(nm) != i: out[i] = _stub(m) changed = True return out, changed @override def wrap_model_call(self, request: ModelRequest, handler: Callable) -> ModelResponse: msgs, changed = self._trim(request.messages) return handler(request.override(messages=msgs)) if changed else handler(request)
📌 原理解析：
主流 LLM（如 OpenAI / DeepSeek）校验消息时，严格要求 `tool_call_id` 与 `ToolMessage` 必须成对存在，直接删除历史 ToolMessage 会导致 API 校验失败崩溃。`TrimReadbacksMiddleware` 巧用“替换为 Stub 存根”的策略：保留原有的 `tool_call_id` 和 `name`，仅仅把过期的几千行 JSON 变成一句话 `[older result omitted...]`。既维护了 API 协议合规，又减少了 **90% 以上** 的无用 Token 消耗！
